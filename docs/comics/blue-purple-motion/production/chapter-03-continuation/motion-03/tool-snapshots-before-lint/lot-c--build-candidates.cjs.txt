async function build16(page){
 const mapDir=path.resolve(value('--map-dir',path.join(root,'art-masters/motion-comics/blue-purple/chapter-03-continuation/motion-03/map')));
 const mapBase=path.join(mapDir,'map-base.png'),mapFinal=path.join(mapDir,'map-final.png');
 assert(fs.existsSync(mapBase)&&fs.existsSync(mapFinal),'La carte relue est requise.');
 const markers=['position','unknown','green'],markerOffsets={position:2.85,unknown:3.25,green:3.65};
 const markerFiles=markers.map(id=>path.join(mapDir,`marker-${id}.png`));
 markerFiles.forEach(file=>assert(fs.existsSync(file),`Overlay repère manquant : ${file}`));
 const result=await page.evaluate(async input=>{
  const load=async src=>{const image=new Image();image.src=src;await image.decode();return image;};
  const [art,original,base,final,letters,...markers]=await Promise.all([input.art,input.master,input.base,input.final,input.letters,...input.markers].map(load));
  const canvas=(w,h)=>{const c=document.createElement('canvas');c.width=w;c.height=h;return c;};
  const encoded=c=>c.toDataURL().split(',')[1];
  // Keep the complete 2:1 map and identical overlay coordinates in the 1000x496 case.
  // This 0.8% vertical adaptation preserves district labels and the outer border.
  const fit=image=>{const c=canvas(1000,496),ctx=c.getContext('2d');ctx.drawImage(image,0,0,1000,496);return c;};
  const city=fit(base),markerCanvases=markers.map(fit),blank=canvas(1000,496),complete=canvas(1000,496);
  // Compose the final from the very same resized RGBA plates used by the player;
  // resizing a pre-flattened image changes antialiasing at marker edges.
  const completeContext=complete.getContext('2d');completeContext.drawImage(city,0,0);
  for(const marker of markerCanvases)completeContext.drawImage(marker,0,0);
  const out=canvas(1024,1536),o=out.getContext('2d');o.drawImage(original,0,0);o.save();o.beginPath();o.rect(12,12,1000,496);o.clip();o.drawImage(complete,12,12);o.drawImage(letters,0,0);o.restore();
  const screen=[251,1183,76,30],states=[];
  for(let step=0;step<=8;step++){
   const c=canvas(screen[2],screen[3]),ctx=c.getContext('2d',{willReadFrequently:true});ctx.drawImage(art,...screen,0,0,screen[2],screen[3]);
   if(step){const pixels=ctx.getImageData(0,0,c.width,c.height),p=pixels.data;
    for(let y=0;y<c.height;y++)for(let x=0;x<c.width;x++){
     const i=(y*c.width+x)*4,r=p[i],g=p[i+1],b=p[i+2];
     // Only warm emissive pixels inside the equipment recess; preserve paint and housing.
     const ellipse=Math.max(0,1-((x-37)/37)**2-((y-13)/15)**2);
     const warm=Math.max(0,Math.min(1,(g-b*1.08)/28))*Math.max(0,Math.min(1,(r-25)/55));
     // The saturated white core belongs to the same lamp but has no warm chroma.
     // Include its tightly feathered emitter rectangle, entirely inside the recess.
     const emitter=Math.max(0,Math.min(1,(x-16)/3,(62-x)/3,(y-7)/2,(21-y)/3));
     const gain=(step/8)*Math.min(1,ellipse*2.8)*Math.max(warm,emitter)*.98;
     p[i]=Math.round(r*(1-gain)+11*gain);p[i+1]=Math.round(g*(1-gain)+12*gain);p[i+2]=Math.round(b*(1-gain)+20*gain);
    }ctx.putImageData(pixels,0,0);
   }states.push(encoded(c));
  }
  const lastScreen=await load('data:image/png;base64,'+states.at(-1));o.drawImage(lastScreen,screen[0],screen[1]);
  return {city:encoded(city),complete:encoded(complete),markers:markerCanvases.map(encoded),blank:encoded(blank),master:encoded(out),screen,screenStates:states};
 },{art:uri(local(data.motionBaseImage)),master:uri(local(data.image)),base:uri(mapBase),final:uri(mapFinal),letters:uri(local(data.letteringImage)),markers:markerFiles.map(uri)});
 const png=(name,bytes)=>{const file=path.join(assets,name);fs.writeFileSync(file,Buffer.from(bytes,'base64'));return media(file);};
 const base=png('night-city-base.png',result.city),final=png('night-city-final.png',result.complete),blank=png('transparent.png',result.blank);
 const overlays=result.markers.map((bytes,i)=>png(`marker-${markers[i]}.png`,bytes));
 data.image=png('page-final.png',result.master);
 const initial=source.motionLayers.find(layer=>layer.mode==='digitalReveal').frames[0].image;
 data.motionLayers=[{id:'elf-world-to-night-city',panelId:'p16-c01',mode:'digitalReveal',rect:[...data.panels[0].rect],startSeconds:hold(0),endSeconds:end(0),
  frames:[{atSeconds:0,image:initial},{atSeconds:2.35,image:base}],reveal:{durationSeconds:1.8,columns:16,rows:8,amplitudePx:13,color:'#55eafa'}},
  ...markers.map((id,i)=>({id:`map-marker-${id}`,panelId:'p16-c01',mode:'frames',rect:[...data.panels[0].rect],startSeconds:hold(0),endSeconds:end(0),frames:[{atSeconds:0,image:blank},{atSeconds:markerOffsets[id],image:overlays[i]}]}))];
 const screenImages=result.screenStates.map((bytes,i)=>png(`screen-${i}.png`,bytes));
 data.motionLayers.push({id:'capsule-screen-extinction',panelId:'p16-c03',mode:'frames',rect:rect(...result.screen),clipPolygon:polygon([[251,1183],[327,1183],[327,1213],[251,1213]]),startSeconds:hold(2),endSeconds:end(2),
  frames:screenImages.map((image,i)=>({atSeconds:i===0?0:.5+i*.16,image}))});
 data.effects=[
  clipped('reflection','p16-c02',[[411,748],[431,706],[489,693],[515,701],[509,749]],{color:'#b65be9',strength:.52,frequency:.38,phase:.8,softEdge:true,clock:'story',startSeconds:hold(1)+.15,endSeconds:hold(1)+2.4,envelope:{attackSeconds:.35,releaseSeconds:.9}}),
  clipped('reflection','p16-c02',[[618,706],[657,712],[644,749],[618,751]],{color:'#52dbed',strength:.38,frequency:.20,phase:.5,softEdge:true,clock:'story',startSeconds:hold(1)+.45,endSeconds:hold(1)+3.1,envelope:{attackSeconds:.5,releaseSeconds:1.2}})
 ];
 report.mapSources=[mapBase,mapFinal,...markerFiles].map(file=>({file,sha256:sha(file)}));
 Object.assign(report.anchors,{vision:hold(0),disassembly:hold(0)+.55,mapStable:hold(0)+2.35,position:hold(0)+2.85,unknown:hold(0)+3.25,green:hold(0)+3.65,glassActivation:hold(1)+.15,glassStable:hold(1)+3.1,screenResidual:hold(2)+.5,screenOff:hold(2)+1.78});
 report.notes.push('Vue subjective: paysage .55 s, désagrégation continue 1.8 s puis trois dossiers fixes successifs: deux jaunes, un vert. Aucun repère HOLT (précision utilisateur).','Vitre: deux polygones hors visage et paroi; retour au dessin stable.','Extinction réelle du petit indicateur chaud dans le renfoncement de la capsule, neuf états localisés; aucun pixel du boîtier déplacé.','OFF: nouvelle carte finale avec trois dossiers et indicateur éteint; master elfique historique conservé.');
 return final;
}
